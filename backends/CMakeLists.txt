# The back ends, one module each (backends/ID/): its code, tests and docs
# (README.md). Each can be left out of the build; register_backends()
# (generated below) registers those built in.
option(BUILD_BACKEND_LOCAL "Local folders" ON)
option(BUILD_BACKEND_SYNCTHING "Syncthing folders" ON)
option(BUILD_BACKEND_CALDAV "CalDAV accounts (needs libcurl and libxml2)" ON)
option(BUILD_BACKEND_WEBDAV "WebDAV folders (needs libcurl and libxml2)" ON)
option(BUILD_BACKEND_GIT "git repositories (runs the git command)" ON)

set(REMINDERS_BACKENDS "")  # ids built in, in the order the Add Source form lists them
set(BACKEND_LIBS "")
find_package(Python3 COMPONENTS Interpreter)

# A test program for a module, with the core's test harness.
function(backend_tests name)
  cmake_parse_arguments(T "" "MAIN" "SOURCES;LIBS;ARGS" ${ARGN})
  if(NOT T_MAIN)
    set(T_MAIN ${PROJECT_SOURCE_DIR}/core/tests/main.cpp)
  endif()
  add_executable(${name} ${T_MAIN} ${T_SOURCES})
  target_include_directories(${name} PRIVATE ${PROJECT_SOURCE_DIR}/core/tests ${CMAKE_CURRENT_SOURCE_DIR}/common/tests)
  target_link_libraries(${name} PRIVATE ${T_LIBS})
  add_test(NAME ${name} COMMAND ${name} ${T_ARGS})
endfunction()

if(BUILD_BACKEND_LOCAL)
  add_library(reminders_backend_local STATIC local/src/local.cpp)
  target_link_libraries(reminders_backend_local PUBLIC reminders_core)
  list(APPEND REMINDERS_BACKENDS local)
  list(APPEND BACKEND_LIBS reminders_backend_local)
endif()

# register_backends(): calls each built-in module's register_ID_backend().
set(DECLS "")
set(CALLS "")
foreach(id IN LISTS REMINDERS_BACKENDS)
  string(APPEND DECLS "void register_${id}_backend();\n")
  string(APPEND CALLS "\t\tregister_${id}_backend();\n")
endforeach()
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/register_backends.cpp.in
"// Generated by backends/CMakeLists.txt: the back ends built in.
#include <mutex>

#include \"reminders/backend_module.hpp\"

namespace rem {

${DECLS}
void register_backends() {
	static std::once_flag once;
	std::call_once(once, [] {
${CALLS}\t});
}

}  // namespace rem
")
configure_file(${CMAKE_CURRENT_BINARY_DIR}/register_backends.cpp.in ${CMAKE_CURRENT_BINARY_DIR}/register_backends.cpp COPYONLY)
add_library(reminders_backends STATIC ${CMAKE_CURRENT_BINARY_DIR}/register_backends.cpp)
target_link_libraries(reminders_backends PUBLIC reminders_core ${BACKEND_LIBS})
message(STATUS "Back ends: ${REMINDERS_BACKENDS}")
